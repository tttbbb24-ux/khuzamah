rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function hasAcc() { return request.auth != null && exists(/databases/$(database)/documents/access/$(request.auth.uid)); }
    function acc() { return get(/databases/$(database)/documents/access/$(request.auth.uid)).data; }
    function ready() { return exists(/databases/$(database)/documents/config/usersReady); }
    function usr() { return get(/databases/$(database)/documents/users/$(acc().user)).data; }
    function named() { return 'user' in acc() && usr().active == true; }
    function member() { return hasAcc() && (!ready() || named()); }
    function partner() { return hasAcc() && (!ready() || (named() && usr().role == 'partner')); }
    function anyStation() { return hasAcc() && ready() && named() && usr().role == 'station'; }
    function stationOwner(st) { return hasAcc() && ready() && named() && usr().role == 'station' && usr().station == st; }

    match /access/{uid} {
      allow read, delete: if request.auth != null && request.auth.uid == uid;
      allow create, update: if request.auth != null && request.auth.uid == uid && (
        (!ready() && request.resource.data.keys().hasOnly(['code'])
          && request.resource.data.code == get(/databases/$(database)/documents/secret/main).data.code)
        || (request.resource.data.keys().hasOnly(['code', 'user']) && request.resource.data.code is string
          && get(/databases/$(database)/documents/codes/$(request.resource.data.code)).data.user == request.resource.data.user)
      );
    }
    match /codes/{code} {
      allow get: if request.auth != null;
      allow list, write: if partner();
    }
    match /users/{id} {
      allow read: if member();
      allow write: if partner();
    }
    match /config/{id} {
      allow read: if member();
      allow create: if partner();
      allow update, delete: if partner() && id != 'usersReady';
    }
    match /seats/{id} {
      allow read: if member();
      allow create: if partner() || (anyStation() && request.resource.data.type == 'take');
      allow update, delete: if partner();
    }
    match /counters/{id} {
      allow read, write: if member();
    }
    match /orders/{id} {
      allow read: if member();
      allow create: if partner() || stationOwner(request.resource.data.st);
      allow delete: if partner() || stationOwner(resource.data.st);
      allow update: if partner() || (stationOwner(resource.data.st)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['qty', 'served', 'servedAt']));
    }
    match /sales/{id} {
      allow read: if member();
      allow create, delete: if partner();
    }
    match /log/{id} {
      allow read: if partner();
      allow create: if member() && (!ready() || request.resource.data.by == acc().user);
    }
  }
}
